// Глобальные объекты, доступные коду в тренажёре (без DOM: код выполняется в Web Worker).
interface Console { log(...data: any[]): void; info(...data: any[]): void; warn(...data: any[]): void; error(...data: any[]): void; table(data: any): void; }
declare var console: Console;
declare function setTimeout(handler: (...args: any[]) => void, timeout?: number, ...args: any[]): number;
declare function clearTimeout(id: number | undefined): void;
declare function setInterval(handler: (...args: any[]) => void, timeout?: number, ...args: any[]): number;
declare function clearInterval(id: number | undefined): void;
declare function queueMicrotask(callback: () => void): void;
declare function structuredClone<T>(value: T): T;
/** Учебная задержка: Promise, который выполнится через ms миллисекунд. */
declare function sleep(ms: number): Promise<void>;
/** Учебный «сервер»: возвращает JSON заказа с задержкой, как настоящий API (данные учебные). */
declare function fakeFetchOrder(id: number): Promise<unknown>;
/** Мини-тесты тренажёра: test('название', () => { ... }) — тело может быть async. */
declare function test(name: string, fn: () => unknown): void;
/** Сравнивает значения по содержимому (как JSON) и падает с сообщением «ожидалось …, получено …». */
declare function expectEq(actual: unknown, expected: unknown, message?: string): void;
/** Падает с сообщением message, если value ложно. */
declare function expectTrue(value: unknown, message: string): void;
/** Ожидает, что fn бросит ошибку (или вернёт отклонённый Promise). Используйте с await. */
declare function expectThrows(fn: () => unknown, message: string): Promise<void>;
/** URL и URLSearchParams (есть в Worker; объявлены здесь, т. к. тренажёр подключает только lib.es2022). */
declare class URLSearchParams { constructor(init?: string | Record<string, string> | [string, string][]); get(name: string): string | null; getAll(name: string): string[]; has(name: string): boolean; set(name: string, value: string): void; append(name: string, value: string): void; delete(name: string): void; toString(): string; forEach(cb: (value: string, key: string) => void): void; entries(): IterableIterator<[string, string]>; keys(): IterableIterator<string>; [Symbol.iterator](): IterableIterator<[string, string]>; }
declare class URL { constructor(url: string, base?: string | URL); href: string; origin: string; protocol: string; host: string; hostname: string; port: string; pathname: string; search: string; hash: string; readonly searchParams: URLSearchParams; toString(): string; }
